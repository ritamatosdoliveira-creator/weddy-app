rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /weddings/{weddingId} {
      // Quem já está autorizado pode ler e editar tudo
      allow read: if request.auth != null &&
        request.auth.token.email in resource.data.authorizedEmails;

      allow update: if request.auth != null && (
        request.auth.token.email in resource.data.authorizedEmails ||
        // ...ou é alguém novo a juntar-se, adicionando só o seu próprio email
        (request.auth.token.email in request.resource.data.authorizedEmails &&
         request.resource.data.authorizedEmails.size() == resource.data.authorizedEmails.size() + 1)
      );

      // Criar um casamento novo: só se o criador se incluir a si próprio
      allow create: if request.auth != null &&
        request.auth.token.email in request.resource.data.authorizedEmails;
    }
  }
}
